# Runtime render settings. Change these without recompiling.
scene = veach-mis
scene_path = ./scenes/
spp = 64
# 0 means auto. Set this lower than your CPU logical thread count to leave room.
threads = 20
stop_rate = 0.6

# Integrator selection.
#   use_mis = 1: MIS path tracer (light + BRDF sampling, power heuristic) - default
#   use_mis = 0: original pre-MIS rendering equation (split diffuse/specular,
#                unweighted NEE + uniform-hemisphere indirect)
use_mis = 0

# Live preview window. Turn it off for maximum render speed.
# Rendering is progressive: 1 sample-per-pixel per pass, up to spp passes,
# so the image starts noisy and refines each pass. The title shows spp N/total.
# preview_interval_ms is how often the window repaints; larger = less slowdown.
# preview_fullscreen opens borderless fullscreen (F11 toggles, Esc exits).
# progressive = 1: refine over time (1spp/pass). = 0: one-shot full spp per pixel
#   (marginally faster to the final image; regions pop in at final quality).
preview = 1
preview_interval_ms = 250
preview_fullscreen = 0
progressive = 1

# Interactive free-fly camera (needs preview = 1). WASD move, Space/Ctrl up-down,
# drag mouse to look, wheel adjusts speed, F11 fullscreen, Esc quit. While moving
# it renders at 1/moving_downscale resolution, then refines to full res once still.
# On quit the current frame is saved to image.png.
interactive = 1
move_speed = 1.0
mouse_sensitivity = 1.0
moving_downscale = 4

# Remove isolated bright fireflies before bilateral filtering.
outlier_removal = true
outlier_radius = 1
outlier_threshold = 2.5

# Post-process denoising. Radius 2 is a conservative start.
bilateral_filter = true
bilateral_radius = 2
bilateral_sigma_space = 2.0
bilateral_sigma_color = 0.12
